assassinate_ruler =  {
	category = covert
	require_acceptance = no
	is_visible = {
		FROM = {
			OR = {
				is_rival = ROOT
				is_enemy = ROOT
				overlord_of = ROOT
				is_subject_of = ROOT
			}
		}
	}
	is_allowed = {
		has_spy_network_in = {
			who = FROM
			value = 50
		}
		custom_trigger_tooltip = {
			tooltip = cannot_assassinate_again_fifty_years
			OR = {
				NOT = { has_country_flag = assassinated_ruler_within_fifty_years }
				had_country_flag = {
					flag = assassinated_ruler_within_fifty_years
					days = 18250
				}
			}
			FROM = {
				OR = {
					NOT = { has_country_flag = had_recent_ruler_assassination }
					had_country_flag = {
						flag = had_recent_ruler_assassination
						days = 18250
					}
				}
			}
		}
		FROM = {
			calc_true_if = {
				amount = 2
				NOT = {
					OR = {
						legitimacy = 50
						horde_unity = 50
						devotion = 50
						republican_tradition = 50
					}
				}
				NOT = {
					stability = 1
				}
				NOT = { any_estate_loyalty = { loyalty = 30 } }
			}
		}
	}
	
	on_accept = {
		set_country_flag = assassinated_ruler_within_fifty_years
		FROM = {
			kill_ruler = yes
			set_country_flag = had_recent_ruler_assassination
		}
		add_spy_network_in = {
			who = FROM
			value = -100
		}
	}
	ai_will_do = {
		OR = {
			FROM = {
				has_reform = monarchy_mechanic
				NOT = { has_heir = yes }
			}
			FROM = {
				adm = 5
				dip = 5
				mil = 5
			}
		}
	}
}

join_league =  {
	category = alliance
	require_acceptance = no
	is_visible = {
		NOT = {
			num_of_cities = 5
		}
		government = republic
		OR = {
			FROM = {
				is_neighbor_of = ROOT
			}
			any_neighbor_country = {
				is_subject_of_type = league_member
				is_subject_of = FROM
			}
		}
	}
	is_allowed = {
		NOT = {
			OR = {
				num_of_cities = FROM
				total_development = FROM
			}
		}
		OR = {
			FROM = {
				is_great_power = yes
			}
			FROM = {
				has_reform = signoria_reform
			}
		}
		OR = {
			AND = {
				has_reform = italian_communes
				FROM = { has_reform = italian_communes }
			}
			AND = {
				has_reform = signoria_reform
				FROM = { has_reform = signoria_reform }
			}
		}
	}
	
	on_accept = {
		FROM = {
			create_subject = {
				subject_type = league_member
				who = ROOT
			}
		}
	}
	ai_will_do = {
		has_opinion = {
			who = FROM
			value = 0
		}
		any_neighbor_country = {
			army_size = ROOT
		}
	}
}

leave_league =  {
	category = alliance
	require_acceptance = no
	is_visible = {
		NOT = {
			num_of_cities = 5
		}
		government = republic
		OR = {
			FROM = {
				is_neighbor_of = ROOT
			}
			any_neighbor_country = {
				is_subject_of_type = league_member
				is_subject_of = FROM
			}
		}
	}
	is_allowed = {
		is_subject_of_type = league_member
	}
	
	on_accept = {
		grant_independence = yes
		FROM = {
			add_opinion = {
				who = ROOT
				modifier = left_league
			}
		}
	}
	ai_will_do = {
		liberty_desire = 50
	}
}
