assassinate_ruler = {
	category = covert
	require_acceptance = no
	is_visible = {
		FROM = {
			OR = {
				is_rival = ROOT
				is_enemy = ROOT
				overlord_of = ROOT
				is_subject_of = ROOT
			}
		}
	}
	is_allowed = {
		has_spy_network_in = {
			who = FROM
			value = 50
		}
		custom_trigger_tooltip = {
			tooltip = cannot_assassinate_again_fifty_years
			OR = {
				NOT = { has_country_flag = assassinated_ruler_within_fifty_years }
				had_country_flag = {
					flag = assassinated_ruler_within_fifty_years
					days = 18250
				}
			}
			FROM = {
				OR = {
					NOT = { has_country_flag = had_recent_ruler_assassination }
					had_country_flag = {
						flag = had_recent_ruler_assassination
						days = 18250
					}
				}
			}
		}
		FROM = {
			calc_true_if = {
				amount = 2
				NOT = {
					OR = {
						legitimacy = 50
						horde_unity = 50
						devotion = 50
						republican_tradition = 50
					}
				}
				NOT = {
					stability = 1
				}
				NOT = { any_estate_loyalty = { loyalty = 30 } }
			}
		}
	}
	
	on_accept = {
		set_country_flag = assassinated_ruler_within_fifty_years
		FROM = {
			kill_ruler = yes
			set_country_flag = had_recent_ruler_assassination
		}
		add_spy_network_in = {
			who = FROM
			value = -100
		}
	}
	ai_will_do = {
		OR = {
			FROM = {
				has_reform = monarchy_mechanic
				NOT = { has_heir = yes }
			}
			FROM = {
				adm = 5
				dip = 5
				mil = 5
			}
		}
	}
}

join_league = {
	category = alliance
	require_acceptance = no
	is_visible = {
		NOT = {
			num_of_cities = 5
		}
		government = republic
		OR = {
			FROM = {
				is_neighbor_of = ROOT
			}
			any_neighbor_country = {
				is_subject_of_type = league_member
				is_subject_of = FROM
			}
		}
	}
	is_allowed = {
		NOT = {
			OR = {
				num_of_cities = FROM
				total_development = FROM
			}
		}
		OR = {
			FROM = {
				is_great_power = yes
			}
			FROM = {
				has_reform = signoria_reform
			}
		}
		OR = {
			AND = {
				has_reform = italian_communes
				FROM = { has_reform = italian_communes }
			}
			AND = {
				has_reform = signoria_reform
				FROM = { has_reform = signoria_reform }
			}
		}
	}
	
	on_accept = {
		FROM = {
			create_subject = {
				subject_type = league_member
				who = ROOT
			}
		}
	}
	ai_will_do = {
		has_opinion = {
			who = FROM
			value = 0
		}
		any_neighbor_country = {
			army_size = ROOT
		}
	}
}

leave_league = {

		}
		government = republic
		OR = {

		liberty_desire = 50
	}
}

war_intervention = {
	category = great_powers
	require_acceptance = no
	is_visible = {
		has_global_flag = concert_of_europe
	}
	is_allowed = {
		FROM = {
			same_continent = ROOT
			OR = {
				is_in_war = {
					defenders = FROM
					duration = 800
				}
				is_in_war = {
					attackers = FROM
					duration = 800
				}
			}
		}
		OR = {
			AND = {
				is_revolutionary = yes
				FROM = {
					is_revolutionary = yes
				}
			}
			AND = {
				is_revolutionary = no
				FROM = {
					is_revolutionary = no
				}
			}
		}
	}
	
	on_accept = {
		join_all_offensive_wars_of = FROM
		join_all_defensive_wars_of = FROM
	}
	ai_will_do = {
		OR = {
			has_opinion = {
				who = FROM
				value = 100
			}
			FROM = {
				any_war_enemy_country = {
					OR = {
						is_rival = ROOT
						is_enemy = ROOT
					}
				}
			}
			FROM = {
				any_neighbor_country = {
					OR = {
						is_rival = ROOT
						is_enemy = ROOT
					}
				}
			}
		}
	}
}

integrate_countries = {
	category = influence
	require_acceptance = yes
	is_visible = {
		has_reform = facilitate_nation_state
		FROM = {
			capital_scope = {
				continent = ROOT
			}
			culture_group = ROOT
		}
	}
	is_allowed = {
		always = yes
	}

	ai_acceptance = {
		add_entry = {
			name = OPINION
			export_to_variable = { which = ai_value value = opinion who = FROM with = ROOT } 
			multiply_variable = { which = ai_value value = 0.3 }
		}

		add_entry = {
			name = ROOT_DIPREP
			export_to_variable = { 
				which = ai_value 
				value = modifier:diplomatic_reputation
			} 
			multiply_variable = { which = ai_value value = 3 } #3
		}

		add_entry = {
			name = TOTAL_DEV_ROOT
			export_to_variable = { 
				which = ai_value 
				value = total_development
			} 
			multiply_variable = { which = ai_value value = 0.05 }
		}

		add_entry = {
			name = BORDER_DISTANCE
			export_to_variable = { 
				which = ai_value 
				value = border_distance 
				who = FROM
			} 
			multiply_variable = { which = ai_value value = -0.1 }
		}

		add_entry = {
			name = TOTAL_DEV
			export_to_variable = { 
				which = ai_value 
				value = total_development 
				who = FROM
			} 
			multiply_variable = { which = ai_value value = -0.05 }
		}

		
		
		add_entry = {
			name = ROOT_FROM_MILITARY_POWER_COMPARED
			limit = {
				variable_arithmetic_trigger = {
					export_to_variable = {
						which = our_mil_tech
						value = trigger_value:mil_tech
					}
					export_to_variable = {
						which = their_mil_tech
						value = trigger_value:mil_tech
						who = FROM
					}
					export_to_variable = {
						which = our_army_size
						value = trigger_value:army_size
					}
					export_to_variable = {
						which = their_army_size
						value = trigger_value:army_size
						who = FROM
					}
					export_to_variable = {
						which = military_power
						value = 0
					}
					divide_variable = { which = our_army_size value = 10 }
					divide_variable = { which = their_army_size value = -10 }
					change_variable = { which = our_army_size which = their_army_size }
					multiply_variable = { which = their_mil_tech value = -1 }
					change_variable = { which = our_mil_tech which = their_mil_tech }
					multiply_variable = { which = our_mil_tech value = 3 }
					change_variable = { which = military_power which = our_army_size }
					change_variable = { which = military_power which = our_mil_tech }
					NOT = { check_variable = { which = military_power value = -39 } }
				}
			}
			export_to_variable = {
				which = ai_value
				value = 40
			}
			multiply_variable = { which = ai_value value = -1.5 }
		}

		add_entry = {
			name = HOSTILE_RELATIONS
			limit = {
				is_enemy = ROOT
			}
			export_to_variable = { which = ai_value value = -1000 }
		}
	}
	
	on_accept = {
		create_subject = {
			subject_type = constituent_state
			who = FROM
		}
	}
	ai_will_do = {
		always = yes
	}
}
