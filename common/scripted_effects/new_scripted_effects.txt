culture_leader_qualification_assessment = {
    add_country_modifier = {
        name = default_culture_leader_modifier
        duration = -1
    }
    custom_tooltip = culture_leader_rewards_vary
    add_treasury = -2000
	if = {
		limit = {
			calc_true_if = {
				amount = 2
				AND = {
					NOT = {
						average_home_autonomy = 5
					}
					stability = 2
				}
				AND = {
					trade_income_percentage = 0.33
          		  	current_income_balance = 2.5
          		  	monthly_income = 50
				}
				AND = {
					has_global_modifier_value = {
						which = improve_relation_modifier
						value = 0.5
					}
					diplomatic_reputation = 5
					variable_arithmetic_trigger = {
						export_to_variable = {
							variable_name = num_allies
							value = trigger_value:num_of_allies
							who = ROOT
						}
						export_to_variable = {
							variable_name = num_subjects
							value = trigger_value:num_of_subjects
							who = ROOT
						}
						change_variable = {
							which = num_allies
							which = num_subjects
						}
						check_variable = {
							which = num_allies
							value = 12
						}
					}
				}
				AND = {
					power_projection = 75
					OR = {
						has_global_modifier_value = {
							which = land_morale
							value = 0.25
						}
						has_global_modifier_value = {
							which = discipline
							value = 0.15
						}
					}
				}
			}
		}
		country_event = {
			id = culture_leader_options.1
		}
	}
    else_if = {
        limit = {
            NOT = {
                average_home_autonomy = 5
            }
            stability = 2
        }
        add_country_modifier = {
            name = economic_culture_leader_modifier
            duration = -1
        }
    }
	else_if = {
        limit = {
            trade_income_percentage = 0.33
            current_income_balance = 2.5
            monthly_income = 50
        }
        add_country_modifier = {
            name = commercial_culture_leader_modifier
            duration = -1
        }
    }
	else_if = {
        limit = {
            has_global_modifier_value = {
                which = improve_relation_modifier
                value = 0.5
            }
            diplomatic_reputation = 5
            variable_arithmetic_trigger = {
                export_to_variable = {
                    variable_name = num_allies
                    value = trigger_value:num_of_allies
                    who = ROOT
                }
                export_to_variable = {
                    variable_name = num_subjects
                    value = trigger_value:num_of_subjects
                    who = ROOT
                }
                change_variable = {
               		which = num_allies
                    which = num_subjects
                }
                check_variable = {
                	which = num_allies
                    value = 12
                }
            }
        }
        add_country_modifier = {
            name = diplomatic_culture_leader_modifier
            duration = -1
        }
    }
	else_if = {
        limit = {
            power_projection = 75
            OR = {
                has_global_modifier_value = {
                    which = land_morale
                    value = 0.25
                }
                has_global_modifier_value = {
                    which = discipline
                    value = 0.15
                }
            }
        }
        add_country_modifier = {
            name = military_culture_leader_modifier
            duration = -1
        }
    }
}
