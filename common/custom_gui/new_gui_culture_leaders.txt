custom_button = {
    name = open_culture_leader_opener_button
    potential = {
        FROM = {
			ai = no
		}
        ROOT = {
            ai = no
        }
    }
    trigger = {
		always = yes
	}
    effect = { 
		if = {
            limit = {
                has_country_flag = opened_culture_leaders_panel
            }
            clr_country_flag = opened_culture_leaders_panel
        }
        else = {
            set_country_flag = opened_culture_leaders_panel
        }
	}
    tooltip = open_culture_leader_tt
}
