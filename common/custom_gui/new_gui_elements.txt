    name = duke_upgrade_button
    potential = {
		always = no

		OR = {

custom_button = {
    name = purchase_prov
    potential = {
        FROM = {
			has_country_flag = opened_prov_dip_panel
			ai = no
		}
    }
    trigger = {
		ROOT = {
			controlled_by = owner
			is_capital = no
			range = FROM
			is_part_of_hre = no
			FROM = { treasury = 100 }
			owner = {
				OR = {
					is_subject = no
					is_subject_of = FROM
				}
				OR = {
					NOT = {
						is_subject_of = FROM
					}
					AND = {
						is_subject_of = FROM
						NOT = { liberty_desire = 50 }
					}
				}
				NOT = {
					tag = FROM
				}
				NOT = {
					truce_with = FROM
				}
				NOT = {
					alliance_with = FROM
				}
			}
		}
	}
    effect = {
		FROM = {
			save_event_target_as = province_purchaser
		}
		ROOT = {
			province_event = {
				id = NEW_DIPLOMACY.1
			}
		}
	}
    tooltip = purchase_province_tooltip
}
