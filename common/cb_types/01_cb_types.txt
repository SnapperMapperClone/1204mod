cb_mongol_enforce_integration = {
	valid_for_subject = no

	prerequisites_self = {
		has_country_flag = assertive_diplomacy_flag
	}

	prerequisites = {
		OR = {
			FROM = {
				is_neighbor_of = ROOT
			}

				is_neighbor_of = FROM
			}
		}
	}

	war_goal = enforce_territorial_integration_wargoal
}
