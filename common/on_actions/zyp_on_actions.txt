on_startup = {
	if = {
		limit = {
			NOT = { has_global_flag = game_started }
		}
		set_global_flag = game_started

		every_province = {
			limit = {
				superregion = china_superregion
				culture_group = east_asian
				OR = {
					has_terrain = farmlands
					has_terrain = mountain
					has_terrain = hills
					has_terrain = jungle
				}
				NOT = {
					area = taiwan_area
				}
			}

			add_province_triggered_modifier = chi_reduced_cost_imperial_modifier
		}
		KHA = {
			save_global_event_target_as = kha_mongol_empire
		}
		REB = {
			save_global_event_target_as = culture_leader_one
			save_global_event_target_as = culture_leader_two
			save_global_event_target_as = culture_leader_three
			save_global_event_target_as = culture_leader_four
			save_global_event_target_as = culture_leader_five
			save_global_event_target_as = culture_leader_six
		}
	}
}

on_main_war_lost = {
	if = {
		limit = {
			NOT = { prestige = 0 }
			check_if_tag_is_culture_leader = yes
		}
		remove_culture_leader = yes
	}
}


on_monarch_death = {
	if = {
		limit = {
			has_country_flag = baldwin_of_flanders
		}
		every_country = {
			limit = {
				has_country_flag = baldwin_of_flanders
			}
			clr_country_flag = baldwin_of_flanders
			kill_ruler = yes
		}
	}

	if = {
		limit = {
			has_country_flag = innocent_iii
		}
		every_country = {
			limit = {
				has_country_flag = innocent_iii
			}
			clr_country_flag = innocent_iii
			kill_ruler = yes
		}
	}
}

on_new_heir = {
	if = {
		limit = {
			has_reform = male_exclusive_inheritance
			has_female_heir = yes
		}
		country_event = {
			id = inheritance_reform_event.1
		}
	}
	if = {
		limit = {
			is_female = yes
			has_foreign_consort = yes
		}
		random_country = {
			limit = {
				ROOT = {
					is_origin_of_consort = PREV
				}
			}
			save_event_target_as = consort_tag
		}
		remove_heir = {
			allow_new_heir = no
		}
		define_heir = {
			age = 0
			male = yes
			dynasty = event_target:origin_of_consort
		}
	}
}

on_create_vassal = {
	ROOT = {
		save_event_target_as = subject_created
	}
	FROM = {
		country_event = {
			id = new_feature_designate_subject.1
		}
	}
}
