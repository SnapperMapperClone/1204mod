on_startup = {
	if = {
		limit = {
			NOT = { has_global_flag = game_started }
		}
		set_global_flag = game_started

		every_province = {
			limit = {
				superregion = china_superregion
				culture_group = east_asian
				OR = {
					has_terrain = farmlands
					has_terrain = mountain
					has_terrain = hills
					has_terrain = jungle
				}
				NOT = {
					area = taiwan_area
				}
			}

			add_province_triggered_modifier = chi_reduced_cost_imperial_modifier
		}
		KHA = {
			save_global_event_target_as = kha_mongol_empire
		}
		REB = {
			save_global_event_target_as = culture_leader_one
			save_global_event_target_as = culture_leader_two
			save_global_event_target_as = culture_leader_three
			save_global_event_target_as = culture_leader_four
			save_global_event_target_as = culture_leader_five
			save_global_event_target_as = culture_leader_six
		}
		every_country = {
			add_yearly_manpower = 5
		}
		every_country = {
			limit = {
				has_heir = yes
			}
			set_heir_flag = is_heir_flag
		}
		every_province = {
			limit = {
				has_port = yes
			}
			add_permanent_province_modifier = {
				name = province_is_coastal
				duration = -1
			}
		}
		every_province = {
			limit = {
				owner = {
					OR = {
						tag = FRA
						is_subject_of = FRA
						is_subject_of = ENG
					}
				}
			}
			set_province_flag = royal_demesne_flag
		}
	}
}

on_siege_won_province = {
	if = {
		limit = {
			is_capital = no
			is_core = controller
			NOT = {
				OR = {
					has_building = fort_15th
					has_building = fort_16th
					has_building = fort_17th
					has_building = fort_18th
				}
			}
			OR = {
				controller = {
					is_in_war = {
						attacker_leader = FROM
						casus_belli = cb_revolution
					}
				}
				controller = {
					is_in_war = {
						defender_leader = FROM
						casus_belli = cb_revolution
					}
				}
			}
			NOT = {
				any_neighbor_province = {
					OR = {
						has_building = fort_15th
						has_building = fort_16th
						has_building = fort_17th
						has_building = fort_18th
					}
					controller = {
						tag = FROM
					}
				}
			}
		}
		cede_province = controller
	}
	else_if = {
		limit = {
			is_capital = no
			is_core = controller
			OR = {
				has_building = fort_15th
				has_building = fort_16th
				has_building = fort_17th
				has_building = fort_18th
			}
			OR = {
				controller = {
					is_in_war = {
						attacker_leader = FROM
						casus_belli = cb_revolution
					}
				}
				controller = {
					is_in_war = {
						defender_leader = FROM
						casus_belli = cb_revolution
					}
				}
			}
		}
		every_province = {
			limit = {
				is_neighbor_of = ROOT
			}
			cede_province = controller
		}
		cede_province = controller
		controller = { add_treasury = 50 }
	}
}

on_new_monarch = {
	if = {
		limit = {
			government = monarchy
			has_regency = no
			NOT = {
				has_ruler_flag = is_heir_flag
			}
			NOT = { is_subject_of_type = appanage }
		}
		country_event = {
			id = new_dynasty_election_event.1
		}
	}

	if = {
		limit = {
			government = monarchy
			has_regency = no
			NOT = {
				has_ruler_flag = is_heir_flag
			}
			is_subject_of_type = appanage
		}
		overlord = {
			create_union = ROOT
		}
	}
}

on_yearly_pulse = {
	if = {
		limit = {
			any_owned_province = {
				devastation = 90
				OR = {
					base_tax = 2
					base_production = 2
					base_manpower = 2
				}
			}
		}
		every_owned_province = {
			limit = {
				devastation = 90
				OR = {
					base_tax = 2
					base_production = 2
					base_manpower = 2
				}
			}
			add_base_tax = -1
			add_base_production = -1
			add_base_manpower = -1
		}
	}
}
