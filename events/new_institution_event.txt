country_event = {
	id = new_institution_events.1
	title = new_institution_events.1.t
	desc = new_institution_events.1.d
	picture = GLOBAL_eventPicture

	is_triggered_only = yes
	
	major = yes

	goto = institution_origin

	trigger = {
		always = yes			#
	}

	option = {
		name = new_institution_events.1.a
		add_adm_power = 100
		add_dip_power = 100
		add_mil_power = 100
		change_innovativeness = 5
		add_prestige = 25
		if = {
			limit = {
				NOT = { has_completed_all_reforms_trigger = yes }
			}
			add_reform_progress_medium_effect = yes
		}
		custom_tooltip = new_institution_events.2.a.tt
		FROM = {
			add_permanent_province_modifier = {
				name = "birthplace_of_ag_rev"
				duration = -1
			}
			hidden_effect = {
				save_global_event_target_as = ag_rev_institution_spawn
			}
		}
		hidden_effect = {
			every_country = {
				limit = {
					ai = no
					knows_country = ROOT
					NOT = { tag = ROOT }
				}
				country_event = { id = new_institution_events.100 }
			}
			every_country = {
				limit = {
					ai = no
					NOT = { knows_country = ROOT }
					NOT = { tag = ROOT }
				}
				country_event = { id = new_institution_events.101 }
			}
		}
	}
}
