#Carthagenian
province_event = {
	id = african_latin.1
	title = african_latin.1.t
	desc = african_latin.1.d
	picture = RELIGIOUS_CONVERSION_eventPicture

	trigger = {
		NOT = {
			culture = carthagenian
		}
		OR = {
			province_id = 352
			area = tunisia_area
			area = djerba_area
		}
		owner = {
			religion_group = christian
		}
		religion_group = christian
	}

	mean_time_to_happen = {
		years = 50
		modifier = {
			factor = 0.5
			any_neighbor_province = { culture_group = african_latin }
		}
		modifier = {
			factor = 0.01
			culture_group = african_latin
		}		
	}

	option = {
		name = african_latin.1.a
		change_culture = carthagenian
	}
}
