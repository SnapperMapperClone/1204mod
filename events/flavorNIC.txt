#Conquest of Constantinople - Trebizond and Epirus still exist
country_event = {
	id = flavor_nic.1
	title = flavor_nic.1.t
	desc = flavor_nic.1.d
	picture = {
		picture = CONSTANTINOPLE_eventPicture
		trigger = {
			has_dlc = "King of Kings"
		}
	}
	picture = {
		picture = CITY_VIEW_eventPicture
		trigger = {
			not = { has_dlc = "King of Kings" }
		}
	}

	trigger = {
		tag = NIC
		owns = 151
		NOT = { exists = BYZ }
		AND = {
			exists = TRE
			exists = EPI
		}
	}

	mean_time_to_happen = {
		months = 12
	}

	option = {
		name = flavor_nic.1.a
		if = {
			limit = {
				exists = TRE
				valid_for_personal_unions_trigger = yes
				TRE = {
		  			valid_for_personal_unions_trigger = yes
			  	}
			  }
			  add_casus_belli = {
				  type = cb_restore_personal_union
				  months = 300
				  target = TRE
			  }
		  }
		  if = {
			  limit = {
				  exists = EPI
				  valid_for_personal_unions_trigger = yes
	  			EPI = {
		  			valid_for_personal_unions_trigger = yes
			  	}
			  }
			  add_casus_belli = {
				  type = cb_restore_personal_union
				  months = 300
				  target = EPI
			}
		}
	}
}

#Conquest of Constantinople - Trebizond and Epirus no longer exist
country_event = {
	id = flavor_nic.4
	title = flavor_nic.1.t
	desc = flavor_nic.4.d
	picture = {
		picture = CONSTANTINOPLE_eventPicture
		trigger = {
			has_dlc = "King of Kings"
		}
	}
	picture = {
		picture = CITY_VIEW_eventPicture
		trigger = {
			not = { has_dlc = "King of Kings" }
		}
	}

	trigger = {
		tag = NIC
		owns = 151
		NOT = { exists = BYZ }
		NOT = {
			AND = {
				exists = TRE
				exists = EPI
			}
		}
	}

	mean_time_to_happen = {
		months = 12
	}

	option = {
		name = flavor_nic.4.a
		change_tag = BYZ
		swap_non_generic_missions = yes
		on_change_tag_effect = yes
		set_government_rank = 3
		151 = {
			move_capital_effect = yes
		}
		hidden_effect = {
			change_primary_culture = greek
			every_province = {
				limit = {
					culture_group = byzantine
				}
				add_core = BYZ
			}
			every_province = {
				limit = {
					OR = {
						region = balkan_region
						region = anatolia_region
					}
				}
				add_permanent_claim = BYZ
			}
		}
	}
}

#Reunite the empire
country_event = {
	id = flavor_nic.5
	title = flavor_nic.1.t
	desc = flavor_nic.5.d
	picture = {
		picture = CONSTANTINOPLE_eventPicture
		trigger = {
			has_dlc = "King of Kings"
		}
	}
	picture = {
		picture = CITY_VIEW_eventPicture
		trigger = {
			not = { has_dlc = "King of Kings" }
		}
	}

	trigger = {
		tag = NIC
		owns = 151
		NOT = { exists = BYZ }
		OR = {
			NOT = { exists = TRE }
			AND = {
				exists = TRE
				TRE = { is_subject_of = ROOT }
			}
		}
		OR = {
			NOT = { exists = EPI }
			AND = {
				exists = EPI
				EPI = { is_subject_of = ROOT }
			}
		}
	}

	mean_time_to_happen = {
		months = 12
	}

	option = {
		name = flavor_nic.4.a
		if = {
			limit = {
				exists = TRE
			}
			inherit = TRE
		}
		if = {
			limit = {
				exists = EPI
			}
			inherit = EPI
		}
		if = {
			limit = {
				AND = {
					exists = MOE
					MOE = { is_subject_of = ROOT }
				}
			}
			inherit = MOE
		}
		change_tag = BYZ
		swap_non_generic_missions = yes
		on_change_tag_effect = yes
		151 = {
			move_capital_effect = yes
		}
		set_government_rank = 3
		hidden_effect = {
			change_primary_culture = greek
			every_province = {
				limit = {
					culture_group = byzantine
				}
				add_core = BYZ
			}
			every_province = {
				limit = {
					OR = {
						region = balkan_region
						region = anatolia_region
					}
				}
				add_permanent_claim = BYZ
			}
		}
	}
}
