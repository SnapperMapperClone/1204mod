country_event = {
	id = flavor_gwy.1
	title = flavor_gwy.1.t
	desc = flavor_gwy.1.d
	picture = DIPLOMACY_eventPicture

	fire_only_once = yes
	major = no
  
	trigger = {
		OR = { 
			tag = GWY
			tag = WLS
			}
		is_year = 1204
		government = monarchy
	}

	mean_time_to_happen = {
		months = 24
	}

  	option = {
		name = "flavor_gwy.1.a"
		if = {
			potential = { ruler = "Llywelyn fawr" }
			NOT = { has_consort = yes }
			define_consort = {
				name = "Joan"
				country_of_origin = ENG
				culture = french
				age = 17
				religion = catholic
				female = yes
				adm = 3
				dip = 5
				mil = 3
			}
			create_marriage = ENG
			add_prestige = 20
		}
		else = {
			add_prestige = 20
			add_stability = 2
		}
	}

	option = {
		name = "flavor_gwy.1.b"
		add_prestige = 10
		add_stability = 1
		add_legitimacy = 20
	}
}
