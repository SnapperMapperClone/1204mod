	title = flavor_wls.1.t

	}

	mean_time_to_happen = {
		months = 24
	}

  	option = {

				name = "Joan"

				culture = french

				mil = 3
			}
			create_marriage = ENG

		}
		else = {
			add_prestige = 20

		}
	}

	option = {
		name = "flavor_wls.1.b"
		add_prestige = 10
		add_stability = 1

	}
