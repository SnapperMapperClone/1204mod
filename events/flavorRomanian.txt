namespace = flavor_romanian

#Romanians in Thracia
province_event = {
	id = flavor_romanian.1
	title = flavor_romanian.1.t
	desc = flavor_romanian.1.d
	picture = RELIGIOUS_CONVERSION_eventPicture

	trigger = {
		NOT = { culture = thracian }
		OR = {
			area = bulgaria_area
			area = silistria_area
		}
		owner = { culture_group = carpathian }
	}

	mean_time_to_happen = {
		years = 40
		modifier = {
			factor = 0.75
			any_neighbor_province = { culture_group = carpathian }
		}		
	}

	option = {
		name = flavor_romanian.1.a
		change_culture = thracian
	}
}
