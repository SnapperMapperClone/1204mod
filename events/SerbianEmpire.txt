namespace = serbian_byz

#Romej culture in greece
province_event = {
	id = serbian_byz.1
	title = serbian_byz.1.t
	desc = serbian_byz.1.d
	picture = STREET_SPEACH_eventPicture

	trigger = {
		owner = { has_country_flag = formed_serbian_byzantium_flag }
		OR = {
			area = hellas_area
			area = albania_area
			area = northern_greece_area
			area = thrace_area
			area = macedonia_area
			area = vardarska_area
		}
		NOT = { culture = romeji }
	}

	mean_time_to_happen = {
		years = 50
		modifier = {
			factor = 0.5
			any_neighbor_province = { culture_group = south_slavic }
		}
	}

	option = {
		name = serbian_byz.1.a
		change_culture = romeji
	}
}
