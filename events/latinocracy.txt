#Latin empire Claims on Greece
country_event = {
	id = latin_greece.3
	title = latin_greece.3.t
	desc = latin_greece.3.d
	picture = {
		picture = CONSTANTINOPLE_eventPicture
		trigger = {
			has_dlc = "King of Kings"
		}
	}
	picture = {
		picture = CITY_VIEW_eventPicture
		trigger = {
			not = { has_dlc = "King of Kings" }
		}
	}

	fire_only_once = yes

	trigger = {
		tag = LAT
		is_year = 1206
		OR = {
			NOT = { TES = { is_subject_of = ROOT } }
			NOT = { ACH = { is_subject_of = ROOT } }
		}
	}

	mean_time_to_happen = {
		months = 4
	}

	option = {
		name = latin_greece.3.a
		if = {
			limit = { NOT = { TES = { is_subject_of = ROOT } } }
			TES = {
				country_event = {
					id = latin_greece.4
					days = 10
				}
			}
		}
		if = {
			limit = { NOT = { ACH = { is_subject_of = ROOT } } }
			ACH = {
				country_event = {
					id = latin_greece.7
					days = 10
				}
			}
		}
	}
}
