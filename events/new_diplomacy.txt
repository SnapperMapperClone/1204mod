province_event = { #confiscate province
	id = new_diplomacy.2
	title = "new_diplomacy.2.t"
	desc = "new_diplomacy.2.d"
	picture = BAD_WITH_MONARCH_eventPicture
	
	is_triggered_only = yes

	option = {
		name = "new_diplomacy.2.a"
		owner = {
			overlord = {
				every_neighbor_country = {
					limit = {
						NOT = {
							is_subject_of = overlord
						}
					}
					add_aggressive_expansion = {
						who = overlord
						value = 2
					}
				}
				every_subject_country = {
					limit = {
						NOT = {
							tag = PREV
						}
					}
					add_liberty_desire = 5
				}
			}
			add_liberty_desire = 10
			add_truce_with = overlord
		}
		remove_core = owner
		owner = {
			overlord = {
				ROOT = {
					cede_province = PREV
				}
			}
		}
	}
}
